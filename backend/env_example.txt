# Supabase Configuration
# Get these values from your Supabase project dashboard

# Project URL (found in Settings > API)
SUPABASE_URL=https://your-project-id.supabase.co

# Anon/Public key (found in Settings > API)
SUPABASE_ANON_KEY=your-anon-key-here

# Service Role key (found in Settings > API) - Keep this secret!
SUPABASE_SERVICE_ROLE_KEY=your-service-role-key-here

# JWT Secret (found in Settings > API > JWT Settings)
SUPABASE_JWT_SECRET=your-jwt-secret-here

# Database Connection (found in Settings > Database)
DATABASE_URL=postgresql://postgres:[YOUR-PASSWORD]@db.[YOUR-PROJECT-ID].supabase.co:5432/postgres

# SendGrid Configuration
# Get these values from your SendGrid account dashboard

# SendGrid API Key (found in Settings > API Keys)
SENDGRID_API_KEY=your-sendgrid-api-key-here

# From Email (verified sender email in SendGrid)
SENDGRID_FROM_EMAIL=noreply@yourdomain.com

# To Email (where contact form emails will be sent)
SENDGRID_TO_EMAIL=contact@yourdomain.com
